import { getToken, AuthUser } from "../hooks/useAuth";
import SurveyCard from "../components/SurveyCard";

interface Message {
  role: "user" | "assistant";
  content: string;
  streaming?: boolean;
}

  id: string;

  user: AuthUser;
  onLogout: () => void;

  const [exchangeCount, setExchangeCount] = useState(0);
  const [sessionClosed, setSessionClosed] = useState(false);
  const [sending, setSending] = useState(false);
  const [error, setError] = useState("");
  const [surveyQuestion, setSurveyQuestion] = useState<SurveyQuestion | null>(null);
  const [surveyDismissed, setSurveyDismissed] = useState(false);
  const bottomRef = useRef<HTMLDivElement>(null);
  const textareaRef = useRef<HTMLTextAreaElement>(null);

  useEffect(() => {
    bottomRef.current?.scrollIntoView({ behavior: "smooth" });
  }, [messages, surveyQuestion, sessionClosed]);

  useEffect(() => {
    createSession();
  }, []);

  const createSession = async () => {
    const token = getToken();
    if (!token) return;
    try {
      const res = await fetch("/api/sessions", {
        method: "POST",
        headers: { "Content-Type": "application/json", Authorization: `Bearer ${token}` },
        body: JSON.stringify({ agentType: "MNT" }),
      });
      if (!res.ok) { setError("Não foi possível iniciar a sessão."); return; }
      const session = await res.json();
      setSessionId(session.id);
      setExchangeCount(session.exchangeCount ?? 0);
    } catch {
      setError("Erro de conexão ao iniciar sessão.");
    }
  };

  const handleNewSession = async () => {
    setMessages([]);
    setExchangeCount(0);
    setSessionClosed(false);
    setSurveyQuestion(null);
    setSurveyDismissed(false);
    setError("");
    await createSession();
  };

  const sendMessage = async (e?: FormEvent) => {
    e?.preventDefault();
    const text = input.trim();
    if (!text || !sessionId || sending || sessionClosed) return;

    const token = getToken();
    if (!token) return;

    setInput("");
    setSending(true);
    setError("");

    if (textareaRef.current) {
      textareaRef.current.style.height = "auto";
    }

    setMessages((prev) => [
      ...prev,
      { role: "user", content: text },
      { role: "assistant", content: "", streaming: true },

    try {
      const res = await fetch("/api/session/message", {
        method: "POST",
        headers: { "Content-Type": "application/json", Authorization: `Bearer ${token}` },
        body: JSON.stringify({ sessionId, content: text }),
      });

      if (!res.ok || !res.body) {
        const data = await res.json().catch(() => ({}));
        setError(data.error ?? "Erro ao enviar mensagem.");
        setMessages((prev) => prev.slice(0, -1));
        setSending(false);
        return;
      }

      const reader = res.body.getReader();
      const decoder = new TextDecoder();
      let buffer = "";

      while (true) {
        const { done, value } = await reader.read();
        if (done) break;

        buffer += decoder.decode(value, { stream: true });

          const raw = line.slice(6).trim();
          if (!raw) continue;

            const event = JSON.parse(raw);

                const updated = [...prev];
                const last = updated[updated.length - 1];
                if (last?.role === "assistant") {
                  updated[updated.length - 1] = { ...last, content: last.content + event.content, streaming: true };
                }
                return updated;
              });
            } else if (event.type === "replace_last") {
              setMessages((prev) => {
                const updated = [...prev];
                const last = updated[updated.length - 1];
                if (last?.role === "assistant") {
                  updated[updated.length - 1] = { ...last, content: event.content, streaming: true };
                }
                return updated;
              });
            } else if (event.type === "done") {
              setExchangeCount(event.exchangeCount);
              setSessionClosed(event.status === "closed");
              setMessages((prev) => {
                const updated = [...prev];
                const last = updated[updated.length - 1];
                if (last?.role === "assistant") {
                  updated[updated.length - 1] = { ...last, streaming: false };
                }
                return updated;
              });
              if (event.surveyQuestion) setSurveyQuestion(event.surveyQuestion);
            } else if (event.type === "error") {
              setError(event.error ?? "Erro no agente.");
              setMessages((prev) => prev.slice(0, -1));
            }
          } catch {
            // ignore parse errors
          }
        }
      }
    } catch {
      setError("Erro de conexão.");
      setMessages((prev) => prev.slice(0, -1));
    }

            <IaiIcon width={28} height={33} />
            <span style={{ color: "#F2F2F2", fontSize: "18px", fontWeight: 700, lineHeight: 1 }}>IAí</span>
          </div>
          <div style={{ color: "#B8953F", fontSize: "10px", letterSpacing: "0.12em", fontWeight: 400 }}>
            by PRIORIS
          </div>
        </div>
        <button
          onClick={onLogout}
          style={{

import { useState, useEffect } from "react";
export interface AuthUser { id: string; email: string; role: string; organizationId: string; organizationName: string; }
const TOKEN_KEY = "iai_auth_token";
const USER_KEY = "iai_auth_user";
const MARKER = "cookie";
function clearStoredUser() { localStorage.removeItem(TOKEN_KEY); localStorage.removeItem(USER_KEY); }
export function useAuth() {
  const [token, setToken] = useState<string | null>(null);
  const [user, setUser] = useState<AuthUser | null>(null);
  const [loading, setLoading] = useState(true);
  const login = (_newToken: string, newUser: AuthUser) => {
    localStorage.setItem(TOKEN_KEY, MARKER); localStorage.setItem(USER_KEY, JSON.stringify(newUser));
    setToken(MARKER); setUser(newUser); setLoading(false);
  };
  const logout = async () => {
    try {
      const res = await fetch("/api/auth/logout", { method: "POST", credentials: "same-origin" });
      if (!res.ok) { window.alert("Não foi possível encerrar sua sessão. Tente novamente."); return false; }
      clearStoredUser(); setToken(null); setUser(null); return true;
    } catch { window.alert("Não foi possível encerrar sua sessão. Verifique sua conexão e tente novamente."); return false; }
  };
  useEffect(() => {
    let cancelled = false;
    clearStoredUser();
    const incoming = new URLSearchParams(window.location.hash.slice(1)).get("token") ?? new URLSearchParams(window.location.search).get("token");
    if (incoming) { setLoading(false); return; }
    fetch("/api/auth/me", { credentials: "same-origin", cache: "no-store" }).then(async res => {
      if (!res.ok) return null;
      const candidate = await res.json();
      return candidate && ["id", "email", "role", "organizationId", "organizationName"].every(key => typeof candidate[key] === "string") ? candidate as AuthUser : null;
    }).then(candidate => {
      if (cancelled) return;
      if (candidate) login(MARKER, candidate); else { setToken(null); setUser(null); }
      setLoading(false);
    }).catch(() => { if (!cancelled) { setToken(null); setUser(null); setLoading(false); } });
    return () => { cancelled = true; };
  }, []);
  return { token, user, loading, login, logout };
}
export function getToken(): string | null { return localStorage.getItem(TOKEN_KEY) === MARKER ? MARKER : null; }

import type { Request, Response, NextFunction } from "express";
import { authRepository, publicUser } from "../lib/auth-repository";
import { credentialMatches, digestToken, isSameOrigin, parseAuthState, requestCredential } from "../lib/auth-state";
export interface AuthenticatedRequest extends Request { user?: { id: string; email: string; role: string; organizationId: string; organizationName: string; }; }
export async function requireAuth(req: AuthenticatedRequest, res: Response, next: NextFunction): Promise<void> {
  const credential = requestCredential(req.headers, req.cookies);
  if (!credential.token) { res.status(401).json({ error: "Entre novamente para continuar." }); return; }
  if (credential.cookie && !["GET", "HEAD", "OPTIONS"].includes(req.method) && !isSameOrigin(req.headers)) { res.status(403).json({ error: "Solicitação não permitida." }); return; }
  try {
    const row = await authRepository.byCredential("session", digestToken(credential.token));
    if (!row || !credentialMatches(parseAuthState(row.authToken), "session", credential.token)) { res.status(401).json({ error: "Sua sessão venceu. Entre novamente." }); return; }
    req.user = publicUser(row); next();
  } catch { res.status(503).json({ error: "Serviço temporariamente indisponível. Tente novamente." }); }
}
